<script>

    // Configurazione Versionamento e UI

    const APP_VERSION = "1.0";

    console.log(`Green Claim Audit Tool inizializzato. Versione: ${APP_VERSION}`);


    // Funzione per ordinare Urgente -> Conforme (Priorità 1 -> 3)

    let isAscending = true;

    function sortTable() {

        const tbody = document.getElementById('tableBody');

        const rows = Array.from(tbody.querySelectorAll('tr'));

        

        rows.sort((a, b) => {

            if (isAscending) {

                return a.dataset.priority - b.dataset.priority;

            } else {

                return b.dataset.priority - a.dataset.priority;

            }

        });

        

        isAscending = !isAscending;

        rows.forEach(row => tbody.appendChild(row));

    }


    // Funzione per filtrare i risultati tramite la Dropdown

    function filterByStatus() {

        const filterValue = document.getElementById('filterStatus').value;

        const rows = document.querySelectorAll('#tableBody tr');

        

        rows.forEach(row => {

            // Estrapola il testo esatto ignorando spazi vuoti

            const status = row.cells[4].innerText.trim();

            if (filterValue === "" || status === filterValue) {

                row.style.display = "";

            } else {

                row.style.display = "none";

            }

        });

    }


    // Resetta la tendina dei filtri

    function resetFilter() {

        document.getElementById('filterStatus').value = "";

        filterByStatus();

    }

</script>
